    ```tsx
    export default function ArtistProfileEdit() {
      const navigation = useNavigation<NavigationProp>();
      const { user } = useAuth();
      const [loading, setLoading] = useState(true);
      const [saving, setSaving] = useState(false);
      const [nomeArtistico, setNomeArtistico] = useState("");
      const [bio, setBio] = useState("");
      const [generos, setGeneros] = useState<string[]>([]);
      const [cacheMin, setCacheMin] = useState("");

      useEffect(() => {
        if (!user?.perfilArtistaId) { setLoading(false); return; }
        artistaPublicoService.getPerfilPublico(user.perfilArtistaId)
          .then(p => {
            setNomeArtistico(p.nome_artistico ?? "");
            setBio(p.biografia ?? "");
            setGeneros(p.generos_musicais ? p.generos_musicais.split(",").map(g => g.trim()) : []);
            setCacheMin(p.cache_minimo ? String(p.cache_minimo) : "");
          })
          .catch(() => {})
          .finally(() => setLoading(false));
      }, [user?.perfilArtistaId]);

      const handleSalvar = async () => {
        if (!user?.perfilArtistaId) return;
        setSaving(true);
        try {
          await api.patch(`/usuarios/perfil-artista/${user.perfilArtistaId}`, {
            nome_artistico: nomeArtistico.trim() || undefined,
            biografia: bio.trim() || undefined,
            generos_musicais: generos.join(", ") || undefined,
            cache_minimo: cacheMin ? Number(cacheMin) : undefined,
          });
          Alert.alert("Sucesso", "Perfil atualizado!");
          navigation.goBack();
        } catch (e: any) {
          Alert.alert("Erro", e?.response?.data?.message || "Não foi possível salvar.");
        } finally {
          setSaving(false);
        }
      };
      // ... render com ScrollView, inputs e botão salvar
    }
    ```
